matOut2midiNotes<-function(midi, figOut){
  
  
  miditime<-cumsum(figOut)-figOut[1]
  midiNotes<-data.frame(time=rep(NA, sum(!is.na(midi))), length=rep(NA, sum(!is.na(midi))),  track=rep(NA, sum(!is.na(midi))), channel=rep(NA, sum(!is.na(midi))),
                        note=rep(NA, sum(!is.na(midi))), notename= rep(NA, sum(!is.na(midi))), velocity=rep(NA, sum(!is.na(midi))))
  
  while (sum(!is.na(midi))>0){
    
    trackBuit<-TRUE
    i.nouTrack<-which(is.na(midiNotes$time))[1]
    ij.1stNote<-which(!is.na(midi), T)[1,]
    midiNotes$note[i.nouTrack]<-midi[ij.1stNote[1],ij.1stNote[2]]
    midiNotes$time[i.nouTrack]<-miditime[ij.1stNote[2]]
    midiNotes$length[i.nouTrack]<-figOut[ij.1stNote[2]]
    midi[ij.1stNote[1],ij.1stNote[2]]<-NA
    
    ii<-ij.1stNote[2]+1
    ##while
    
    for (i.col in ii:length(midi[1,])){
      i.aqNote<-which(is.na(midiNotes$note))[1]-1
      
      i.notes<-which(midi[,i.col]==midiNotes$note[i.aqNote])
      
      if(length(i.notes)==0){
        
        i.notes<-which(!is.na(midi[,i.col]))
        if (length(i.notes)>0){
          midiNotes$note[i.aqNote+1]<-midi[i.notes[1],i.col]
          midiNotes$time[i.aqNote+1]<-miditime[i.col]
          midiNotes$length[i.aqNote+1]<-figOut[i.col]
          
        }
      } else {
        
        midiNotes$length[i.aqNote]<-midiNotes$length[i.aqNote]+figOut[i.col]}
      midi[i.notes[1],i.col]<-NA
    }
    print(midi)
  }#while
  
  return(midiNotes)
  
  
})

conv2formaNormal<-function(acord){
acord<-sort(unique(acord%%12))

if(length(acord)==1){return(list(0, acord))}
if( length(unique(diff(c(acord, acord[1]+12))))==1 ){ return(list(acord, acord)) }

intervals_inversions<-matrix(rep(0, length(acord)*length(acord)), ncol=length(acord), nrow=length(acord) )
aux_index_baix_formaNormal<-1;

for (ii in 1:length(acord)){
aux_acord<-sort((acord-acord[ii])%%12)
#print(aux_acord)

intervals_inversions[min(2, ii),]<-c(tail(aux_acord,1) - aux_acord[1], diff(aux_acord) )
#print(intervals_inversions)

iii<-1; 

while (ii>1 & iii <length(intervals_inversions[1,])){
#print(iii)
if(intervals_inversions[1,iii]==intervals_inversions[2,iii]){} 
	else {
		if(intervals_inversions[1,iii]<intervals_inversions[2,iii]){iii<-iii+12} 
			else{intervals_inversions[1,]<-intervals_inversions[2,]; aux_index_baix_formaNormal<-ii} } 
iii<-iii+1;
#print(intervals_inversions)
} #while
} #for
return(list( sort((acord-acord[aux_index_baix_formaNormal])%%12), (sort((acord-acord[aux_index_baix_formaNormal])%%12)+acord[aux_index_baix_formaNormal])%%12  ))
}

arranja_darrer_onset<-function(cohestil_onset, midiFiles){
if(tail(cohestil_onset, 1)==0){ 
	last_cohestil_onset<-cohestil_onset[length(cohestil_onset)-1]/480
	penult_cohestil_onset<-cohestil_onset[length(cohestil_onset)-2]/480
	aux_cohestil<-getMelody(fName=midiFiles)

	cohestil_onset[length(cohestil_onset)]<-cohestil_onset[length(cohestil_onset)-1]
	cohestil_onset[length(cohestil_onset)-1]<-median(480*aux_cohestil$t_onset[aux_cohestil$t_onset>penult_cohestil_onset & aux_cohestil$t_onset<last_cohestil_onset  ])
 }
return(cohestil_onset)
